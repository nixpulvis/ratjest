16/// Starts following the wheel and touches over the view drawn in `screen`,
17/// with cells `cell` CSS pixels wide and tall, scrolling `scroll`. Its area
18/// is the whole view's, as many cells as fit in `screen`, like the frame's
19/// when it's drawn.
20///
21/// The view takes these events over from the browser's own scrolling,
22/// setting `screen`'s `touch-action` to `none` so the browser doesn't pan or
23/// zoom with its touches first, and follows them for as long as the page
24/// lives, so call it once.
25///
26/// With [Ratzilla], what's scrolled is shared between drawing and
27/// scrolling, and the backend measures its cells:
28///
29/// ```ignore
30/// use std::{cell::RefCell, rc::Rc};
31///
32/// use ratjest::{Axis, Scroll, Step};
33/// use ratzilla::ratatui::Terminal;
34/// use ratzilla::ratatui::layout::Rect;
35/// use ratzilla::ratatui::widgets::Paragraph;
36/// use ratzilla::{CellSized, DomBackend, WebRenderer};
37///
38/// /// Some text, and the line at the top of the view.
39/// struct Text {
40///     lines: Vec<String>,
41///     top: usize,
42/// }
43///
44/// impl Scroll for Text {
45///     fn scroll(&mut self, step: Step, area: Rect) -> bool {
46///         let last = self.lines.len().saturating_sub(usize::from(area.height));
47///         step.is_over(area) && step.scroll(Axis::Vertical, &mut self.top, last)
48///     }
49/// }
50///
51/// let text = Rc::new(RefCell::new(Text { lines, top: 0 }));
52/// let terminal = Terminal::new(DomBackend::new_by_element(screen.clone())?)?;
53///
54/// let (width, height) = terminal.backend().cell_size_css_px();
55/// let cell = (f64::from(width), f64::from(height));
56/// ratjest::web::listen(&screen, cell, text.clone())?;
57///
58/// terminal.draw_web(move |frame| {
59///     let text = text.borrow();
60///     let lines = Paragraph::new(text.lines.join("\n")).scroll((text.top as u16, 0));
61///     frame.render_widget(lines, frame.area());
62/// });
63/// ```
64///
65/// [Ratzilla]: https://github.com/ratatui/ratzilla
66pub fn listen(
67    screen: &Element,
68    cell: (f64, f64),
69    scroll: impl Scroll + 'static,
70) -> Result<(), JsValue> {
71    if let Some(screen) = screen.dyn_ref::<HtmlElement>() {
72        screen.style().set_property("touch-action", "none")?;
73    }
74    let listener = Rc::new(RefCell::new(Listener {
75        gestures: Gestures::new(cell.0, cell.1),
76        scroll: Box::new(scroll),
77        screen: screen.clone(),
78        cell,
79        gliding: false,
80    }));
81    listen_wheel(&listener)?;
82    listen_touches(&listener)
83}
84
85type Shared = Rc<RefCell<Listener>>;
86
87struct Listener {
88    gestures: Gestures,
89    scroll: Box<dyn Scroll>,
90    screen: Element,
91    /// How wide and tall a cell is, in pixels.
92    cell: (f64, f64),
93    /// Whether an animation frame is waiting to keep a flick going.
94    gliding: bool,
95}
96
97impl Listener {
98    /// The view's area, as many cells as fit in the screen.
99    fn area(&self) -> Rect {
100        let rect = self.screen.get_bounding_client_rect();
101        let cells = |pixels: f64, size: f64| (pixels / size).max(0.0) as u16;
102        Rect::new(
103            0,
104            0,
105            cells(rect.width(), self.cell.0),
106            cells(rect.height(), self.cell.1),
107        )
108    }
109
110    /// Where `x`, `y` pixels in the window are in the view. The part of a
111    /// cell left over at the screen's far edges counts as its last cell, so
112    /// a gesture there is still over the view.
113    fn in_view(&self, x: i32, y: i32) -> (f64, f64) {
114        let rect = self.screen.get_bounding_client_rect();
115        let area = self.area();
116        let last = |cells: u16, size: f64| (f64::from(cells) * size - 1.0).max(0.0);
117        (
118            (f64::from(x) - rect.left()).min(last(area.width, self.cell.0)),
119            (f64::from(y) - rect.top()).min(last(area.height, self.cell.1)),
120        )
121    }
122
123    /// How far a wheel event goes, sideways and down, in pixels.
124    fn wheel_pixels(&self, e: &WheelEvent) -> (f64, f64) {
125        let pixels = |delta: f64| match e.delta_mode() {
126            WheelEvent::DOM_DELTA_LINE => delta * self.cell.1,
127            WheelEvent::DOM_DELTA_PAGE => delta * f64::from(self.screen.client_height()),
128            _ => delta,
129        };
130        (pixels(e.delta_x()), pixels(e.delta_y()))
131    }
132}

134fn listen_wheel(listener: &Shared) -> Result<(), JsValue> {
135    let shared = listener.clone();
136    let on_wheel = Closure::<dyn FnMut(WheelEvent)>::new(move |e: WheelEvent| {
137        e.prevent_default();
138        let listener = &mut *shared.borrow_mut();
139        let at = listener.in_view(e.client_x(), e.client_y());
140        let delta = listener.wheel_pixels(&e);
141        let area = listener.area();
142        let scroll = &mut *listener.scroll;
143        listener
144            .gestures
145            .wheel(at, delta, e.time_stamp(), scroll, area);
146    });
147    listener
148        .borrow()
149        .screen
150        .add_event_listener_with_callback_and_add_event_listener_options(
151            "wheel",
152            on_wheel.as_ref().unchecked_ref(),
153            &not_passive(),
154        )?;
155    on_wheel.forget();
156    Ok(())
157}

159/// A finger drags the view along, and a flick keeps it gliding.
160fn listen_touches(listener: &Shared) -> Result<(), JsValue> {
161    let shared = listener.clone();
162    let on_move = Closure::<dyn FnMut(TouchEvent)>::new(move |e: TouchEvent| {
163        e.prevent_default();
164        let listener = &mut *shared.borrow_mut();
165        let Some(at) = first_touch(listener, &e) else {
166            return;
167        };
168        let area = listener.area();
169        let scroll = &mut *listener.scroll;
170        listener
171            .gestures
172            .touch_move(at, e.time_stamp(), scroll, area);
173    })
174    .into_js_value();
175
176    let shared = listener.clone();
177    let on_end = Closure::<dyn FnMut(TouchEvent)>::new(move |_: TouchEvent| {
178        let mut listener = shared.borrow_mut();
179        if listener.gestures.touch_end() && !listener.gliding {
180            listener.gliding = true;
181            drop(listener);
182            glide(shared.clone());
183        }
184    })
185    .into_js_value();
186
187    // A touch's events all go to the element it started on, even once a
188    // redraw has taken that element out of the page, when they no longer
189    // bubble up to the view. So the rest of the touch is followed on the
190    // element itself, the innermost one, which may be inside a shadow root.
191    let shared = listener.clone();
192    let on_start = Closure::<dyn FnMut(TouchEvent)>::new(move |e: TouchEvent| {
193        {
194            let listener = &mut *shared.borrow_mut();
195            if let Some(at) = first_touch(listener, &e) {
196                listener.gestures.touch_start(at, e.time_stamp());
197            }
198        }
199        let Ok(target) = e.composed_path().get(0).dyn_into::<EventTarget>() else {
200            return;
201        };
202        // Adding the same listener to an element again does nothing.
203        let _ = target.add_event_listener_with_callback_and_add_event_listener_options(
204            "touchmove",
205            on_move.unchecked_ref(),
206            &not_passive(),
207        );
208        for kind in ["touchend", "touchcancel"] {
209            let _ = target.add_event_listener_with_callback(kind, on_end.unchecked_ref());
210        }
211    });
212    listener
213        .borrow()
214        .screen
215        .add_event_listener_with_callback("touchstart", on_start.as_ref().unchecked_ref())?;
216    on_start.forget();
217    Ok(())
218}

220/// Keeps a flick going a frame at a time, until it stops.
221fn glide(shared: Shared) {
222    let Some(window) = web_sys::window() else {
223        shared.borrow_mut().gliding = false;
224        return;
225    };
226    let looping = shared.clone();
227    let frame = Closure::once_into_js(move |now: f64| {
228        let more = {
229            let listener = &mut *looping.borrow_mut();
230            let area = listener.area();
231            let scroll = &mut *listener.scroll;
232            let more = listener.gestures.glide(now, scroll, area);
233            listener.gliding = more;
234            more
235        };
236        if more {
237            glide(looping);
238        }
239    });
240    if window
241        .request_animation_frame(frame.unchecked_ref())
242        .is_err()
243    {
244        shared.borrow_mut().gliding = false;
245    }
246}

248/// Where the first finger of a touch is in the view, in pixels.
249fn first_touch(listener: &Listener, e: &TouchEvent) -> Option<(f64, f64)> {
250    let touch = e.touches().get(0)?;
251    Some(listener.in_view(touch.client_x(), touch.client_y()))
252}

254/// Listener options that let the view take the event over from the
255/// browser's own scrolling.
256fn not_passive() -> AddEventListenerOptions {
257    let options = AddEventListenerOptions::new();
258    options.set_passive(false);
259    options
